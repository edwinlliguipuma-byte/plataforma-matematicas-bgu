import glob
import pandas as pd
from django.core.management.base import BaseCommand
from django.contrib.auth.models import User
from cursos.models import CursoBGU, EstudiantePerfil


class Command(BaseCommand):
    help = 'Importa masivamente los estudiantes de forma instantÃ¡nea'

    def handle(self, *args, **kwargs):
        mapeo_cursos = {
            "2DO DE BACHILLERATO, TÃ©cnico - Contabilidad A": "2do BGU Contabilidad 'A'",
            "2DO DE BACHILLERATO, TÃ©cnico - Contabilidad B": "2do BGU Contabilidad 'B'",
            "2DO DE BACHILLERATO, TÃ©cnico - InformÃ¡tica A": "2do BGU InformÃ¡tica 'A'",
            "2DO DE BACHILLERATO, TÃ©cnico - InformÃ¡tica B": "2do BGU InformÃ¡tica 'B'",
            "2DO DE BACHILLERATO, TÃ©cnico - InformÃ¡tica C": "2do BGU InformÃ¡tica 'C'",
            "3RO DE BACHILLERATO, TÃ©cnico - Contabilidad A": "3ro BGU Contabilidad 'A'",
            "3RO DE BACHILLERATO, TÃ©cnico - Contabilidad B": "3ro BGU Contabilidad 'B'",
            "3RO DE BACHILLERATO, TÃ©cnico - InformÃ¡tica A": "3ro BGU InformÃ¡tica 'A'",
            "3RO DE BACHILLERATO, TÃ©cnico - InformÃ¡tica B": "3ro BGU InformÃ¡tica 'B'",
            "3RO DE BACHILLERATO, TÃ©cnico - InformÃ¡tica C": "3ro BGU InformÃ¡tica 'C'",
        }

        archivos = glob.glob("*.csv")
        if not archivos:
            self.stdout.write(self.style.WARNING("No se encontraron archivos *.csv en la carpeta actual."))
            return

        total_importados = 0

        for archivo_csv in archivos:
            curso_db_nombre = None
            for clave, valor in mapeo_cursos.items():
                if clave in archivo_csv:
                    curso_db_nombre = valor
                    break

            if not curso_db_nombre:
                self.stdout.write(self.style.WARNING(f"Sin mapeo de curso para: {archivo_csv}"))
                continue

            curso_obj, _ = CursoBGU.objects.get_or_create(nombre=curso_db_nombre)

            df = pd.read_csv(archivo_csv)

            fila_inicio = None
            for idx, row in df.iterrows():
                row_str = str(row.values)
                if 'CÃ‰DULA' in row_str or 'CEDULA' in row_str:
                    fila_inicio = idx + 1
                    break

            if fila_inicio is None:
                self.stdout.write(self.style.WARNING(f"No se encontrÃ³ cabecera CÃ‰DULA/CEDULA en: {archivo_csv}"))
                continue

            importados_archivo = 0
            for idx in range(fila_inicio, len(df)):
                row = df.iloc[idx]
                cedula = str(row.iloc[3]).strip()
                nombres_completos = str(row.iloc[5]).strip()
                correo = str(row.iloc[9]).strip()

                if not cedula or cedula.lower() == 'nan' or len(cedula) < 5:
                    continue

                partes = nombres_completos.split()
                if len(partes) >= 2:
                    apellidos = f"{partes[0]} {partes[1]}"
                    nombres = " ".join(partes[2:]) if len(partes) > 2 else partes[1]
                else:
                    apellidos = nombres_completos
                    nombres = "Estudiante"

                user, creado = User.objects.get_or_create(
                    username=cedula,
                    defaults={
                        'first_name': nombres,
                        'last_name': apellidos,
                        'email': correo if '@' in correo else f"{cedula}@estudiantes2.edu.ec"
                    }
                )
                if creado:
                    user.password = f"pbkdf2_sha256$600000$dummy$fastpass={cedula}"
                    user.save()

                perfil, _ = EstudiantePerfil.objects.get_or_create(
                    user=user,
                    defaults={'cedula': cedula, 'curso': curso_obj}
                )
                perfil.curso = curso_obj
                perfil.cedula = cedula
                perfil.save()

                total_importados += 1
                importados_archivo += 1

            self.stdout.write(f"  - {archivo_csv} -> {curso_db_nombre}: {importados_archivo} estudiantes")

        self.stdout.write(self.style.SUCCESS(f"Â¡EXITO_TOTAL: {total_importados} estudiantes importados perfectamente!"))

import os
import glob
import pandas as pd
from django.core.management.base import BaseCommand
from django.contrib.auth.models import User
from cursos.models import CursoBGU, EstudiantePerfil

class Command(BaseCommand):
    help = 'Importa masivamente los estudiantes de todos los archivos CSV de los 10 cursos'

    def handle(self, *args, **kwargs):
        # Mapeo de nombres de archivos CSV a los nombres exactos de los cursos en la Base de Datos
        mapeo_cursos = {
            "2DO DE BACHILLERATO, TÃ©cnico - Contabilidad A": "2do BGU Contabilidad 'A'",
            "2DO DE BACHILLERATO, TÃ©cnico - Contabilidad B": "2do BGU Contabilidad 'B'",
            "2DO DE BACHILLERATO, TÃ©cnico - InformÃ¡tica A": "2do BGU InformÃ¡tica 'A'",
            "2DO DE BACHILLERATO, TÃ©cnico - InformÃ¡tica B": "2do BGU InformÃ¡tica 'B'",
            "2DO DE BACHILLERATO, TÃ©cnico - InformÃ¡tica C": "2do BGU InformÃ¡tica 'C'",
            "3RO DE BACHILLERATO, TÃ©cnico - Contabilidad A": "3ro BGU Contabilidad 'A'",
            "3RO DE BACHILLERATO, TÃ©cnico - Contabilidad B": "3ro BGU Contabilidad 'B'",
            "3RO DE BACHILLERATO, TÃ©cnico - InformÃ¡tica A": "3ro BGU InformÃ¡tica 'A'",
            "3RO DE BACHILLERATO, TÃ©cnico - InformÃ¡tica B": "3ro BGU InformÃ¡tica 'B'",
            "3RO DE BACHILLERATO, TÃ©cnico - InformÃ¡tica C": "3ro BGU InformÃ¡tica 'C'",
        }

        total_importados = 0

        for archivo_csv in glob.glob("*.csv"):
            # Identificar a quÃ© curso pertenece el archivo por su nombre
            curso_db_nombre = None
            for clave, valor in mapeo_cursos.items():
                if clave in archivo_csv:
                    curso_db_nombre = valor
                    break
            
            if not curso_db_nombre:
                self.stdout.write(self.style.WARNING(f"No se pudo mapear el archivo: {archivo_csv}"))
                continue

            try:
                curso_obj = CursoBGU.objects.get(nombre=curso_db_nombre)
            except CursoBGU.DoesNotExist:
                self.stdout.write(self.style.ERROR(f"El curso '{curso_db_nombre}' no existe en la base de datos."))
                continue

            # Leer el CSV usando pandas (buscando la fila donde empieza la cÃ©dula)
            df = pd.read_csv(archivo_csv)
            
            # Buscamos la fila de inicio de datos
            fila_inicio = None
            for idx, row in df.iterrows():
                row_str = str(row.values)
                if 'CÃ‰DULA' in row_str or 'CEDULA' in row_str:
                    fila_inicio = idx + 1
                    break

            if fila_inicio is None:
                self.stdout.write(self.style.WARNING(f"No se encontrÃ³ la cabecera en {archivo_csv}"))
                continue

            # Procesamos las filas de estudiantes
            for idx in range(fila_inicio, len(df)):
                row = df.iloc[idx]
                cedula = str(row.iloc[3]).strip()
                nombres_completos = str(row.iloc[5]).strip()
                correo = str(row.iloc[9]).strip()

                # Validar que la cÃ©dula sea vÃ¡lida (ej. que no sea NaN o vacÃ­a)
                if not cedula or cedula.lower() == 'nan' or len(cedula) < 5:
                    continue

                # Dividir nombres y apellidos (aproximaciÃ³n: dos primeros nombres / dos apellidos)
                partes = nombres_completos.split()
                if len(partes) >= 2:
                    apellidos = f"{partes[0]} {partes[1]}"
                    nombres = " ".join(partes[2:]) if len(partes) > 2 else partes[1]
                else:
                    apellidos = nombres_completos
                    nombres = "Estudiante"

                # Verificar si el usuario ya existe
                if User.objects.filter(username=cedula).exists():
                    continue

                # Crear usuario de Django (cÃ©dula como username y contraseÃ±a inicial)
                user = User.objects.create_user(
                    username=cedula,
                    email=correo if '@' in correo else f"{cedula}@estudiantes2.edu.ec",
                    password=cedula,
                    first_name=nombres,
                    last_name=apellidos
                )

                # Crear perfil de estudiante vinculado al curso
                EstudiantePerfil.objects.create(
                    user=user,
                    curso=curso_obj,
                    cedula=cedula
                )
                total_importados += 1

        self.stdout.write(self.style.SUCCESS(f"Â¡Se importaron exitosamente {total_importados} estudiantes de todos los cursos!"))

def _normalizar_curso_bgu_post(request):
    if getattr(request, 'method', '') != 'POST':
        return
    raw = (request.POST.get('curso') or '').strip()
    if raw not in ('2DO BGU', '3ERO DE BGU', '2DO DE BGU'):
        return
    import re as _re
    pref = '2DO' if raw.startswith('2') else '3RO'
    esp = (request.POST.get('especialidad') or '').strip().lower()
    par = _re.sub(r'^paralelo\s+', '', (request.POST.get('paralelo') or '').strip().lower()).strip()
    cands = [
        c for c in CursoBGU.objects.all().order_by('id')
        if (_re.match(r'^(2do|2Â°|segundo)\b', (c.nombre or '').strip().lower()) if pref == '2DO'
            else _re.match(r'^(3ro|3ero|3Â°|tercero)\b', (c.nombre or '').strip().lower()))
    ]
    chosen = None
    for c in cands:
        cn = c.nombre.lower()
        par_ok = bool(par and _re.search(r'(?:^|[\s\'"])' + _re.escape(par) + r'(?:$|[\s\'"])', cn))
        if (esp and esp in cn) and par_ok:
            chosen = c
            break
    if not chosen and esp:
        for c in cands:
            if esp in c.nombre.lower():
                chosen = c
                break
    if not chosen and cands:
        chosen = cands[0]
    if chosen:
        mut = request.POST.copy()
        mut['curso'] = chosen.nombre
        request.POST = mut

        if max(img.size) > 1024:
            img.thumbnail((1024, 1024), Image.Resampling.LANCZOS)
        buffer = io.BytesIO()
        img.save(buffer, format="JPEG", quality=85)
        return base64.b64encode(buffer.getvalue()).decode("utf-8")

    def save(self, *args, **kwargs):
        if self.imagen_sello:
            try:
                self.sello_base64 = self._convertir_a_base64(self.imagen_sello)

        if self.imagen_firma:

                self.firma_mime = "image/jpeg"

def _normalizar_curso_bgu_post(request):
    if getattr(request, 'method', '') != 'POST':
        return
    raw = (request.POST.get('curso') or '').strip()
    if raw not in ('2DO BGU', '3ERO DE BGU', '2DO DE BGU'):
        return
    import re as _re
    pref = '2DO' if raw.startswith('2') else '3RO'
    esp = (request.POST.get('especialidad') or '').strip().lower()
    par = _re.sub(r'^paralelo\s+', '', (request.POST.get('paralelo') or '').strip().lower()).strip()
    cands = [
        c for c in CursoBGU.objects.all().order_by('id')
        if (_re.match(r'^(2do|2Â°|segundo)\b', (c.nombre or '').strip().lower()) if pref == '2DO'
            else _re.match(r'^(3ro|3ero|3Â°|tercero)\b', (c.nombre or '').strip().lower()))
    ]
    chosen = None
    for c in cands:
        cn = c.nombre.lower()
        par_ok = bool(par and _re.search(r'(?:^|[\s\'"])' + _re.escape(par) + r'(?:$|[\s\'"])', cn))
        if (esp and esp in cn) and par_ok:
            chosen = c
            break
    if not chosen and esp:
        for c in cands:
            if esp in c.nombre.lower():
                chosen = c
                break
    if not chosen and cands:
        chosen = cands[0]
    if chosen:
        mut = request.POST.copy()
        mut['curso'] = chosen.nombre
        request.POST = mut

def _detectar_especialidad_paralelo(nombre_curso):
    texto = (nombre_curso or "").lower()
    esp = "InformÃ¡tica"
    if "contab" in texto:
        esp = "Contabilidad"
    elif "inform" in texto:
        esp = "InformÃ¡tica"
    elif "venta" in texto:
        esp = "Ventas"
    elif "agrop" in texto:
        esp = "Agropecuaria"

    par = "A"
    for letra in ["A", "B", "C"]:
        if (
            f"'{letra.lower()}'" in texto
            or f'"{letra.lower()}"' in texto
            or texto.endswith(f" {letra.lower()}")
        ):
            par = letra
            break
    return esp, par


def _formatear_whatsapp_ec(telefono):
    limpio = "".join(c for c in (telefono or "") if c.isdigit())
    if not limpio:
        return ""
    if limpio.startswith("0") and len(limpio) == 10:
        return "593" + limpio[1:]
    if limpio.startswith("593"):
        return limpio
    return limpio


def _gemini_urlopen(req, timeout=165):
    import time, urllib.request, urllib.error
    orig_url = req.full_url if hasattr(req, "full_url") else str(req)
    if "generativelanguage.googleapis.com" not in orig_url or ":generateContent" not in orig_url:
        return urllib.request.urlopen(req, timeout=timeout)
    base_part, q_part = orig_url.split("?", 1) if "?" in orig_url else (orig_url, "")
    k = ""
    for seg in q_part.split("&"):
        if seg.startswith("key="):
            k = seg[4:]
    cands = [orig_url]
    if k:
        for alt in ("models/gemini-3.8-flash", "models/gemini-3.5-flash", "models/gemini-flash-latest", "models/gemini-3.0-flash"):
            u_alt = f"https://generativelanguage.googleapis.com/v1beta/{alt}:generateContent?key={k}"
            if u_alt not in cands:
                cands.append(u_alt)
    last_err = None
    for u_try in cands[:3]:
        r_try = urllib.request.Request(u_try, data=req.data, headers=dict(req.header_items()), method=req.get_method())
        try:
            return urllib.request.urlopen(r_try, timeout=165)
        except urllib.error.HTTPError as he:
            last_err = he
            if he.code in (404, 429, 500, 502, 503, 504):
                time.sleep(1.0)
                continue
            raise
    if last_err:
        raise last_err
    return urllib.request.urlopen(req, timeout=165)

def _gemini_url(k):
    import json, urllib.request
    k = (k or "").strip().strip('"').strip("'")
    pref = [
        "models/gemini-3.8-flash",
        "models/gemini-3.5-flash",
        "models/gemini-3.0-flash",
        "models/gemini-flash-latest",
        "models/gemini-3.8-pro",
        "models/gemini-pro-latest",
    ]
    try:
        u = f"https://generativelanguage.googleapis.com/v1beta/models?key={k}"
        d = json.loads(urllib.request.urlopen(u, timeout=165).read().decode("utf-8"))
        ms = [
            m["name"] for m in d.get("models", [])
            if "generateContent" in m.get("supportedGenerationMethods", [])
            and "gemini" in m.get("name", "").lower()
            and not any(x in m.get("name", "").lower() for x in ("tts", "image", "audio", "embedding", "exp", "preview", "1.5", "2.0", "2.5", "vision", "aqa", "veo"))
        ]
        for m in pref:
            if m in ms:
                return f"https://generativelanguage.googleapis.com/v1beta/{m}:generateContent?key={k}"
        fl = sorted([m for m in ms if "flash" in m.lower()], reverse=True)
        if fl:
            return f"https://generativelanguage.googleapis.com/v1beta/{fl[0]}:generateContent?key={k}"
        if ms:
            return f"https://generativelanguage.googleapis.com/v1beta/{sorted(ms, reverse=True)[0]}:generateContent?key={k}"
    except Exception:
        pass
    return f"https://generativelanguage.googleapis.com/v1beta/models/gemini-3.8-flash:generateContent?key={k}"

@staff_member_required
def panel_docente_institucional(request):
    cursos = list(CursoBGU.objects.all().order_by("nombre"))
    curso_id = request.GET.get("curso_id", "")
    exportar = request.GET.get("exportar", "")

    estudiantes_qs = EstudiantePerfil.objects.select_related("curso", "user").order_by(
        "curso__nombre", "user__last_name", "apellidos_nombres"
    )
    if curso_id:
        estudiantes_qs = estudiantes_qs.filter(curso_id=curso_id)

    estudiantes = list(estudiantes_qs)
    est_ids = [e.id for e in estudiantes]

    prom_entregas = defaultdict(dict)
    for row in (
        EntregaTarea.objects.filter(estudiante_id__in=est_ids)
        .values("estudiante_id", "tarea__tipo_tarea")
        .annotate(p=Avg("nota_calculada"))
    ):
        prom_entregas[row["estudiante_id"]][row["tarea__tipo_tarea"]] = float(row["p"] or 0)

    prom_evals_map = {}
    for row in (
        ResultadoEvaluacion.objects.filter(estudiante_id__in=est_ids)
        .values("estudiante_id")
        .annotate(p=Avg("nota_obtenida"))
    ):
        prom_evals_map[row["estudiante_id"]] = float(row["p"] or 0)

    tareas_por_curso = defaultdict(int)
    for t in Tarea.objects.all().values("id", "curso_id"):
        tareas_por_curso[t["curso_id"]] += 1

    evals_por_curso = defaultdict(int)
    for ev in Evaluacion.objects.all().values("id", "curso_id"):
        evals_por_curso[ev["curso_id"]] += 1

    tareas_ok_por_est = defaultdict(set)
    for row in EntregaTarea.objects.filter(
        estudiante_id__in=est_ids, nota_calculada__gte=7
    ).values("estudiante_id", "tarea_id"):
        tareas_ok_por_est[row["estudiante_id"]].add(row["tarea_id"])

    evals_ok_por_est = defaultdict(set)
    for row in ResultadoEvaluacion.objects.filter(
        estudiante_id__in=est_ids, nota_obtenida__gte=7
    ).values("estudiante_id", "evaluacion_id"):
        evals_ok_por_est[row["estudiante_id"]].add(row["evaluacion_id"])

    filas = []
    total_con_pendientes = 0

    for est in estudiantes:
        p_mat = prom_entregas[est.id].get("MATERIA", 0.0)
        p_deb = prom_entregas[est.id].get("DEBER", 0.0)
        p_tal = prom_entregas[est.id].get("TALLER", 0.0)
        p_eva = prom_evals_map.get(est.id, 0.0)

        notas_activas = [n for n in (p_mat, p_deb, p_tal, p_eva) if n > 0]
        promedio_final = round(sum(notas_activas) / len(notas_activas), 2) if notas_activas else 0.0

        tot_t = tareas_por_curso.get(est.curso_id, 0)
        ok_t = len(tareas_ok_por_est.get(est.id, set()))
        tot_e = evals_por_curso.get(est.curso_id, 0)
        ok_e = len(evals_ok_por_est.get(est.id, set()))
        pendientes = max(0, (tot_t - ok_t) + (tot_e - ok_e))
        if pendientes > 0:
            total_con_pendientes += 1

        filas.append(
            {
                "estudiante": est,
                "materia": round(p_mat, 2),
                "deberes": round(p_deb, 2),
                "talleres": round(p_tal, 2),
                "evaluaciones": round(p_eva, 2),
                "promedio": promedio_final,
                "pendientes": pendientes,
            }
        )

    if exportar == "csv":
        response = HttpResponse(content_type="text/csv; charset=utf-8-sig")
        response["Content-Disposition"] = 'attachment; filename="Calificaciones_UE_Enrique_Noboa.csv"'
        writer = csv.writer(response, delimiter=";")
        writer.writerow(['UNIDAD EDUCATIVA TÃ‰CNICA "DR. ENRIQUE NOBOA" - LA TRONCAL'])
        writer.writerow(
            [
                "CÃ©dula",
                "Apellidos y Nombres",
                "Curso",
                "Materia",
                "Deberes",
                "Talleres",
                "Lecciones/Eval.",
                "Promedio General",
                "Actividades Pendientes",
            ]
        )
        for f in filas:
            writer.writerow(
                [
                    f["estudiante"].cedula,
                    f["estudiante"].nombre_completo,
                    f["estudiante"].curso.nombre,
                    f["materia"],
                    f["deberes"],
                    f["talleres"],
                    f["evaluaciones"],
                    f["promedio"],
                    f["pendientes"],
                ]
            )
        return response

    try:
        total_modulos = ModuloInteractivo.objects.count()
    except Exception:
        total_modulos = 0

    return render(
        request,
        "cursos/panel_docente.html",
        {
            "cursos": cursos,
            "curso_id_actual": curso_id,
            "filas": filas,
            "total_estudiantes": len(filas),
            "total_modulos": total_modulos,
            "total_con_pendientes": total_con_pendientes,
        },
    )

def _normalizar_curso_bgu_post(request):
    if getattr(request, 'method', '') != 'POST':
        return
    raw = (request.POST.get('curso') or '').strip()
    if raw not in ('2DO BGU', '3ERO DE BGU', '2DO DE BGU'):
        return
    import re as _re
    pref = '2DO' if raw.startswith('2') else '3RO'
    esp = (request.POST.get('especialidad') or '').strip().lower()
    par = _re.sub(r'^paralelo\s+', '', (request.POST.get('paralelo') or '').strip().lower()).strip()
    cands = [
        c for c in CursoBGU.objects.all().order_by('id')
        if (_re.match(r'^(2do|2Â°|segundo)\b', (c.nombre or '').strip().lower()) if pref == '2DO'
            else _re.match(r'^(3ro|3ero|3Â°|tercero)\b', (c.nombre or '').strip().lower()))
    ]
    chosen = None
    for c in cands:
        cn = c.nombre.lower()
        par_ok = bool(par and _re.search(r'(?:^|[\s\'"])' + _re.escape(par) + r'(?:$|[\s\'"])', cn))
        if (esp and esp in cn) and par_ok:
            chosen = c
            break
    if not chosen and esp:
        for c in cands:
            if esp in c.nombre.lower():
                chosen = c
                break
    if not chosen and cands:
        chosen = cands[0]
    if chosen:
        mut = request.POST.copy()
        mut['curso'] = chosen.nombre
        request.POST = mut

def _clave_alfabetica(estudiante):
    texto = (estudiante.nombre_completo or '').strip().upper()
    return ''.join(
        c for c in unicodedata.normalize('NFD', texto)
        if unicodedata.category(c) != 'Mn'
    )


def _detectar_mime(archivo):
    content_type = getattr(archivo, 'content_type', '') or ''
    if content_type.startswith('image/') or content_type == 'application/pdf':
        return content_type
    nombre = (archivo.name or '').lower()
    if nombre.endswith('.pdf'):
        return 'application/pdf'
    if nombre.endswith('.png'):
        return 'image/png'
    if nombre.endswith('.webp'):
        return 'image/webp'
    return 'image/jpeg'

                if estado_mat not in ('completo', 'incompleto'):
                    estado_mat = 'completo' if nota_final >= 8.0 else 'incompleto'

@csrf_exempt
def api_guardar_nota_evaluacion(request):
    if request.method == 'OPTIONS':
        return _agregar_cors(JsonResponse({'ok': True}))

    if request.method != 'POST':
        return _agregar_cors(JsonResponse({'ok': False, 'error': 'MÃ©todo no permitido'}, status=405))

    try:
        datos = json.loads(request.body.decode('utf-8'))
        cedula = str(datos.get('cedula', '')).strip()
        evaluacion_id = datos.get('evaluacion_id')
        nota = float(str(datos.get('nota', 0)).replace(',', '.'))
        nota = round(max(0.0, min(10.0, nota)), 2)

        if not cedula or not evaluacion_id:
            return _agregar_cors(JsonResponse({'ok': False, 'error': 'Faltan datos del estudiante o mÃ³dulo'}, status=400))

        estudiante = EstudiantePerfil.objects.select_related('user', 'curso').get(cedula=cedula)
        evaluacion = Evaluacion.objects.select_related('curso').get(id=evaluacion_id)

        if estudiante.curso_id != evaluacion.curso_id:
            return _agregar_cors(JsonResponse({
                'ok': False,
                'error': f'Acceso denegado: este mÃ³dulo es de {evaluacion.curso.nombre}, pero estÃ¡s registrado en {estudiante.curso.nombre}.'
            }, status=403))

        ResultadoEvaluacion.objects.update_or_create(
            evaluacion=evaluacion,
            estudiante=estudiante,
            defaults={'nota_obtenida': nota}
        )

        return _agregar_cors(JsonResponse({
            'ok': True,
            'estudiante': estudiante.nombre_completo,
            'nota': nota,
        }))
    except EstudiantePerfil.DoesNotExist:
        return _agregar_cors(JsonResponse({'ok': False, 'error': 'CÃ©dula no encontrada en la plataforma'}, status=404))
    except Evaluacion.DoesNotExist:
        return _agregar_cors(JsonResponse({'ok': False, 'error': 'MÃ³dulo no encontrado'}, status=404))
    except Exception as e:
        return _agregar_cors(JsonResponse({'ok': False, 'error': str(e)}, status=400))

def subir_tarea(request, tarea_id):
    tarea = get_object_or_404(Tarea.objects.select_related('curso'), id=tarea_id)
    tipo = (tarea.tipo_tarea or '').lower().strip()
    error = None
    cedula_inicial = (request.POST.get('cedula') or request.GET.get('cedula') or '').strip()
    tiene_muestra_sello = MuestraSelloFirma.objects.filter(activo=True).exists()

    if request.method == 'POST':
        _normalizar_curso_bgu_post(request)
        cedula = cedula_inicial
        archivos = request.FILES.getlist('archivos_estudiante')
        if not archivos and request.FILES.get('archivo_estudiante'):
            archivos = [request.FILES.get('archivo_estudiante')]

        try:
            estudiante = EstudiantePerfil.objects.select_related('user', 'curso').get(cedula=cedula)

            if estudiante.curso_id != tarea.curso_id:
                error = f"Esta actividad pertenece a {tarea.curso.nombre}, pero tu cÃ©dula estÃ¡ registrada en {estudiante.curso.nombre}."

                sellos_completos = analisis['sellos_completos']
                sellos_atrasados = analisis['sellos_atrasados']
                observacion_ia = analisis['observacion']

                sellos_deb = sellos_completos if tipo == 'deber' else 0
                sellos_tall = sellos_completos if tipo == 'taller' else 0

                archivo_principal = archivos[0]
                archivo_principal.seek(0)

                EntregaTarea.objects.update_or_create(
                    tarea=tarea,
                    estudiante=estudiante,
                    defaults={
                        'archivo_estudiante': archivo_principal,
                        'estado_materia': estado_materia,
                        'sellos_deberes': sellos_deb,
                        'sellos_talleres': sellos_tall,
                        'sellos_atrasados': sellos_atrasados,

                    }
                )
                return render(request, 'cursos/tarea_exitosa.html', {
                    'tarea': tarea,
                    'tipo_tarea': tipo,
                    'estudiante': estudiante,
                    'nota': nota,
                    'estado_materia': estado_materia,
                    'sellos_deberes': sellos_deb,
                    'sellos_talleres': sellos_tall,
                    'sellos_atrasados': sellos_atrasados,
                    'observacion_ia': observacion_ia,
                    'total_archivos': len(archivos),
                })

        except EstudiantePerfil.DoesNotExist:
            error = "La cÃ©dula ingresada no estÃ¡ registrada en el sistema."

    return render(request, 'cursos/subir_tarea.html', {
        'tarea': tarea,
        'tipo_tarea': tipo,
        'error': error,
        'cedula_inicial': cedula_inicial,
        'tiene_muestra_sello': tiene_muestra_sello,
    })

            for t in tareas:
                tareas_con_nota.append({
                    'tarea': t,

                })

    })
